<script>
    // ========== 请替换为你的 Supabase 配置 ==========
    const SUPABASE_URL = 'https://你的项目.supabase.co';
    const SUPABASE_ANON_KEY = '你的anon公钥';
    // =============================================
    const supabase = window.supabase.createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

    let currentUser = null;
    let games = [];
    let userLikes = new Set();

    async function loadGames() {
        const { data, error } = await supabase.from('games').select('*').order('id');
        if (error) { console.error(error); return; }
        games = data.map(g => ({
            id: g.id,
            title: g.title,
            icon_emoji: g.icon_emoji || '🎮',
            desc: g.description,
            file_path: g.file_path,
            heat: 0
        }));
        const likeCounts = await fetchLikeCounts();
        games.forEach(g => { g.heat = likeCounts[g.id] || 0; });
        await fetchUserLikes();
        renderAll();
    }

    async function fetchLikeCounts() {
        const { data, error } = await supabase.from('likes').select('game_id');
        if (error) return {};
        const map = {};
        data.forEach(l => { map[l.game_id] = (map[l.game_id] || 0) + 1; });
        return map;
    }

    async function fetchUserLikes() {
        if (!currentUser) { userLikes.clear(); return; }
        const { data, error } = await supabase.from('likes').select('game_id').eq('user_id', currentUser.id);
        if (!error) {
            userLikes.clear();
            data.forEach(l => userLikes.add(l.game_id));
        }
    }

    async function toggleLike(gameId, btnElement) {
        if (!currentUser) { alert('请先登录'); showLoginModal(); return; }
        const isLiked = userLikes.has(gameId);
        if (isLiked) {
            const { error } = await supabase.from('likes').delete().eq('user_id', currentUser.id).eq('game_id', gameId);
            if (!error) {
                userLikes.delete(gameId);
                btnElement.classList.remove('liked');
                btnElement.innerHTML = '<i class="far fa-heart"></i> 点赞';
                await supabase.rpc('add_points', { p_user_id: currentUser.id, p_points: -1 });
                updateGameHeat(gameId, -1);
            }
        } else {
            const { error } = await supabase.from('likes').insert({ user_id: currentUser.id, game_id: gameId });
            if (!error) {
                userLikes.add(gameId);
                btnElement.classList.add('liked');
                btnElement.innerHTML = '<i class="fas fa-heart"></i> 已赞';
                await supabase.rpc('add_points', { p_user_id: currentUser.id, p_points: 1 });
                updateGameHeat(gameId, 1);
            }
        }
    }

    function updateGameHeat(gameId, delta) {
        const game = games.find(g => g.id === gameId);
        if (game) game.heat = (game.heat || 0) + delta;
        renderAll();
    }

    function renderAll() {
        const hotSorted = [...games].sort((a,b)=>b.heat - a.heat).slice(0,4);
        document.getElementById('hotGameGrid').innerHTML = hotSorted.map(g => renderGameCard(g)).join('');
        document.getElementById('allGameGrid').innerHTML = games.map(g => renderGameCard(g)).join('');
        const trending = [...games].sort((a,b)=>b.heat - a.heat).slice(0,5);
        document.getElementById('trendingList').innerHTML = trending.map(g => `<div class="trend-item" data-id="${g.id}"><span>${escapeHtml(g.title)}</span><span class="trend-score">🔥 ${g.heat}</span></div>`).join('');
        attachLikeEvents();
        attachCardClickEvents();
        attachTrendingClickEvents();
    }

    function escapeHtml(str) { return str.replace(/[&<>]/g, function(m){ if(m==='&') return '&amp;'; if(m==='<') return '&lt;'; if(m==='>') return '&gt;'; return m;}); }

    function renderGameCard(game) {
        const isLiked = userLikes.has(game.id);
        const likeBtnHtml = isLiked ? '<i class="fas fa-heart"></i> 已赞' : '<i class="far fa-heart"></i> 点赞';
        return `
            <div class="game-card" data-id="${game.id}" data-path="${game.file_path || '#'}">
                <div class="card-img">${game.icon_emoji}</div>
                <div class="card-info">
                    <div class="card-title">${escapeHtml(game.title)}</div>
                    <div class="card-desc">${escapeHtml((game.desc || '').substring(0, 60))}</div>
                    <div class="meta-row">
                        <span class="free-tag">免费</span>
                        <button class="like-btn ${isLiked ? 'liked' : ''}" data-id="${game.id}">${likeBtnHtml}</button>
                    </div>
                </div>
            </div>
        `;
    }

    function attachLikeEvents() {
        document.querySelectorAll('.like-btn').forEach(btn => {
            btn.removeEventListener('click', likeHandler);
            btn.addEventListener('click', likeHandler);
        });
    }
    function likeHandler(e) {
        e.stopPropagation();
        const gameId = parseInt(this.dataset.id);
        toggleLike(gameId, this);
    }

    function attachCardClickEvents() {
        document.querySelectorAll('.game-card').forEach(card => {
            card.removeEventListener('click', cardClickHandler);
            card.addEventListener('click', cardClickHandler);
        });
    }
    function cardClickHandler(e) {
        if (e.target.classList.contains('like-btn')) return;
        const gameId = this.dataset.id;
        window.location.href = `game.html?id=${gameId}`;
    }

    function attachTrendingClickEvents() {
        document.querySelectorAll('.trend-item').forEach(item => {
            item.removeEventListener('click', trendingClickHandler);
            item.addEventListener('click', trendingClickHandler);
        });
    }
    function trendingClickHandler(e) {
        const gameId = this.dataset.id;
        if (gameId) window.location.href = `game.html?id=${gameId}`;
    }

    async function loadUpcoming() {
        const { data, error } = await supabase.from('upcoming_games').select('*').order('release_date', { ascending: true }).limit(5);
        if (error) return;
        const html = data.map(g => `<div class="upcoming-item"><span>${escapeHtml(g.title)}</span><span class="upcoming-date">${g.release_date || '待定'}</span></div>`).join('');
        document.getElementById('upcomingList').innerHTML = html || '暂无预告';
    }

    async function loadActiveUsers() {
        const { data, error } = await supabase.from('user_activity').select('user_id, total_points').order('total_points', { ascending: false }).limit(5);
        if (error) return;
        const html = data.map(u => `<div class="active-item"><span>用户 ${u.user_id.slice(0,8)}...</span><span>🔥 ${u.total_points || 0} 分</span></div>`).join('');
        document.getElementById('activeUsersList').innerHTML = html;
    }

    function showLoginModal() {
        const email = prompt('请输入邮箱地址：');
        if (!email) return;
        const pwd = prompt('请输入密码（至少6位）：');
        if (!pwd || pwd.length < 6) return alert('密码至少6位');
        supabase.auth.signUp({ email, password: pwd })
            .then(({ error }) => {
                if (error && error.message.includes('already registered')) {
                    return supabase.auth.signInWithPassword({ email, password: pwd });
                } else if (error) {
                    alert('注册失败：' + error.message);
                    return null;
                } else {
                    alert('注册成功，请登录');
                    return supabase.auth.signInWithPassword({ email, password: pwd });
                }
            })
            .then((result) => {
                if (result && result.error) alert('登录失败：' + result.error.message);
                else if (result && result.data?.user) {
                    currentUser = result.data.user;
                    document.querySelector('.login-btn').innerText = currentUser.email.slice(0,10)+'…';
                    fetchUserLikes().then(() => renderAll());
                }
            });
    }

    window.onload = () => {
        document.getElementById('loginBtn').onclick = showLoginModal;
        loadGames();
        loadUpcoming();
        loadActiveUsers();
        supabase.auth.onAuthStateChange((_, session) => {
            currentUser = session?.user || null;
            if (currentUser) {
                document.querySelector('.login-btn').innerText = currentUser.email.slice(0,10)+'…';
                fetchUserLikes().then(() => renderAll());
            } else {
                document.querySelector('.login-btn').innerText = '登录';
                userLikes.clear();
                renderAll();
            }
        });
        document.getElementById('heroExploreBtn').onclick = () => document.querySelector('.section-head').scrollIntoView({ behavior: 'smooth' });
        document.getElementById('moreHotLink').onclick = () => window.location.href = '#allGameGrid';
        document.getElementById('moreAllLink').onclick = () => window.location.href = '#allGameGrid';
    };
</script>
