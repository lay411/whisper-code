        container.innerHTML = hotSorted.map(game => `

    function renderAllGames() {

    function attachCardEvents() {
        document.querySelectorAll('.game-card').forEach(card => {
            card.addEventListener('click', (e) => {
                const id = parseInt(card.getAttribute('data-id'));
                openGameModal(id);
            });
        });
    }

    function renderTrendingSidebar() {
        const sorted = [...gamesLibrary].sort((a,b)=>b.heat - a.heat).slice(0,5);
        const container = document.getElementById('trendingList');
        container.innerHTML = sorted.map(g => `<div class="trend-item"><span><i class="fas ${g.icon}"></i> ${g.title}</span><span class="trend-score">🔥 ${g.heat}</span></div>`).join('');
    }

    let currentGame = null;
    function openGameModal(gameId) {
        const game = gamesLibrary.find(g => g.id === gameId);
        if(!game) return;
        currentGame = game;
        document.getElementById('modalGameTitle').innerText = game.title;
        document.getElementById('modalFullTitle').innerText = game.title;
        document.getElementById('modalFullDesc').innerHTML = game.desc;
        document.getElementById('modalRuleWarning').innerHTML = `<i class="fas fa-eye-slash"></i> ${game.rule}`;
        document.getElementById('modalIcon').innerHTML = `<i class="fas ${game.icon}"></i>`;
        const giftContainer = document.getElementById('modalGiftList');
        const gifts = game.gifts || ["幽蓝夜灯","暗夜萤火"];
        giftContainer.innerHTML = gifts.map(g => {
            const price = giftPrice[g] || 8;
            return `<div class="gift-item" data-gift="${g}" data-price="${price}"><i class="fas fa-gift"></i> ${g} ¥${price}</div>`;
        }).join('');
        document.querySelectorAll('.gift-item').forEach(el => {
            el.addEventListener('click', function() {
                document.querySelectorAll('.gift-item').forEach(i=>i.classList.remove('active'));
                this.classList.add('active');
                document.getElementById('customAmount').value = '';
            });
        });
        document.getElementById('gameModal').style.display = 'flex';
    }

    function closeModal() { document.getElementById('gameModal').style.display = 'none'; }

    // 打赏逻辑 完全自愿
    function doReward() {
        let amount = 0;
        let selectedGift = null;
        const activeGift = document.querySelector('.gift-item.active');
        const customVal = document.getElementById('customAmount').value;
        if(activeGift) {
            selectedGift = activeGift.getAttribute('data-gift');
            amount = parseFloat(activeGift.getAttribute('data-price'));
        } else if(customVal && !isNaN(parseFloat(customVal)) && parseFloat(customVal) > 0) {
            amount = parseFloat(customVal);
            selectedGift = `自定义打赏 ¥${amount}`;
        } else {
            alert('请选择礼物或输入自定义金额 (自愿支持)');
            return;
        }
        alert(`🎁 感谢您的回声馈赠！\n游戏：《${currentGame.title}》\n礼物/金额：${selectedGift} ¥${amount.toFixed(2)}\n您的支持将全部用于怪谈创作者。`);
        document.querySelectorAll('.gift-item').forEach(el=>el.classList.remove('active'));
        document.getElementById('customAmount').value = '';
    }
